# Helpers: session state init
# =========================
def _init_state():
    defaults = {
        "authed": False,
        "projects": ["datasci-prod", "datasci-dev"],
        "datasets": {
            "datasci-prod": ["sales", "inventory"],
            "datasci-dev": ["staging", "sandbox"],
        },
        # Keyed by (project, dataset)
        "tables": {
            ("datasci-prod", "sales"): ["orders", "customers", "transactions"],
            ("datasci-prod", "inventory"): ["stocks", "shipments"],
            ("datasci-dev", "staging"): ["orders_stg", "customers_stg"],
            ("datasci-dev", "sandbox"): ["demo_table"],
        },
        "selected_project": None,
        "selected_dataset": None,
        "selected_table": None,
        "messages": [],  # chat history: list of {role, content}
    }
    for k, v in defaults.items():
        st.session_state.setdefault(k, v)


def _reset_downstream(after: str):
    """If a higher-level selection changes, clear dependent selections."""
    if after == "project":
        st.session_state.selected_dataset = None
        st.session_state.selected_table = None
    elif after == "dataset":
        st.session_state.selected_table = None


_init_state()

# =========================
# Tab 1: Set Context (auth + selectors)
# =========================
with tab1:
    st.subheader("1) Connect your warehouse")

    c1, c2 = st.columns([2, 5], vertical_alignment="center")
    with c1:
        # Placeholder auth: toggles True on click. Replace with real OAuth later.
        if not st.session_state.authed:
            if st.button("Sign in with Google", use_container_width=True):
                st.session_state.authed = True
        else:
            st.success("Authenticated.")
    with c2:
        if not st.session_state.authed:
            st.warning("Google Auth is required to access your data warehouse.")
        else:
            st.success("Logged in. Fetching your projects, datasets, and tables… ")

    if st.session_state.authed:
        st.divider()
        st.subheader("2) Choose data")

        projects: List[str] = st.session_state.projects
        datasets_by_proj: Dict[str, List[str]] = st.session_state.datasets
        tables_by_ds: Dict[Tuple[str, str], List[str]] = st.session_state.tables

        # Project selector
        proj = st.selectbox(
            "GCP Project",
            options=projects,
            index=(projects.index(st.session_state.selected_project)
                   if st.session_state.selected_project in projects else 0),
            key="_proj_sel",
            help="Pick the GCP project to work in.",
        )
        if proj != st.session_state.selected_project:
            st.session_state.selected_project = proj
            _reset_downstream("project")

        # Dataset selector (depends on project)
        ds_options = datasets_by_proj.get(st.session_state.selected_project, [])
        ds_index = ds_options.index(st.session_state.selected_dataset) if st.session_state.selected_dataset in ds_options else 0
        ds = st.selectbox(
            "Dataset",
            options=ds_options or ["-"],
            index=ds_index if ds_options else 0,
            key="_ds_sel",
            help="Pick a dataset inside the project.",
        )
        if ds_options and ds != st.session_state.selected_dataset:
            st.session_state.selected_dataset = ds
            _reset_downstream("dataset")

        # Table selector (depends on dataset)
        tbl_options = tables_by_ds.get((st.session_state.selected_project, st.session_state.selected_dataset), [])
        tbl_index = tbl_options.index(st.session_state.selected_table) if st.session_state.selected_table in tbl_options else 0
        tbl = st.selectbox(
            "Table",
            options=tbl_options or ["-"],
            index=tbl_index if tbl_options else 0,
            key="_tbl_sel",
            help="Pick a table to analyze.",
        )
        if tbl_options and tbl != st.session_state.selected_table:
            st.session_state.selected_table = tbl

        st.caption(
            f"Active: **{st.session_state.selected_project or '—'}** · "
            f"**{st.session_state.selected_dataset or '—'}** · "
            f"**{st.session_state.selected_table or '—'}"
        )

        st.session_state.messages.append({"role": "assistant", "content": pseudo_sql})
